// 检查猜测
int check_guess(int guess,int TARGET) {
    const int CORRECT = 1, WRONG = 0;
    int result,game_over;
    char hint;
    if (guess == TARGET) {
        result = CORRECT;
        game_over = 1;
    } else {
        result = WRONG;
        if (guess < TARGET) hint = 'L';
        else hint = 'H';
    }
    return result;
}
