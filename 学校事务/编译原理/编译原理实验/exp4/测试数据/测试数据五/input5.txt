// 全局常量
const int MAX_N = 10, ERROR_VAL = 1;

// 全局变量
char algorithm = 'R';      // R:递归, I:迭代

// 函数原型
int fact_rec(int n);
int fact_iter(int n);
void print_fact(int n);
void empty_fact();

// 递归阶乘
int fact_rec(int n) {
    const int BASE = 1;
    int result;
    if (n <= 1) {
        result = 1;
    } else {
        result = n * fact_rec(n - 1);
    }
    return result;
}

// 迭代阶乘
int fact_iter(int n) {
    int result = 1, i;
    i = 2;
    while (i <= n) {
        result = result * i;
        i = i + 1;
    }
    return result;
}

void print_fact(int n) {
    int i = 0;
    do {
        int val;
        if (algorithm == 'R') val = fact_rec(i);
        else val = fact_iter(i);
        i = i + 1;
        if (val > MAX_N) continue;
        if (i > n) break;
    } while (i <= n);
}
